 private static Review deserializeReview(String jsonString)
 {
     if(jsonString != null) {
         //System.out.println("JSON File Content:\n" + jsonString);
         // Convert the JSON string into an object
         Gson gson = new Gson();                                 
         review = gson.fromJson(jsonString, Review.class);
         //System.out.println("Object created from JSON: " + filter);
     } else {
         System.out.println("Failed to read the JSON file.");
     }
     return review;
 }

  private synchronized void giveReview(String roomForReview, double review) {
        boolean roomFound = false;
        for (Room room : rooms) {
            if (room.getRoomName().equals(roomForReview)) {
                int numberof = room.getReviews() + 1;
                double starof = room.getStars();
                double sum = starof * room.getReviews();
                room.setReviews(numberof);
                room.setStars((sum + review) / numberof);
                System.out.println("Review added for room: " + room.getRoomName()+ ", Current stars: " + room.getStars());
                roomFound = true;
            }
        }
        if (!roomFound) {
            System.out.println("No room with the name '" + roomForReview + "' found.");
    }
